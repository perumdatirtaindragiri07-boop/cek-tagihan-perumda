SQL Setup Tabel & Kebijakan Supabase

Salin dan jalankan perintah SQL berikut pada menu SQL Editor di Dashboard Supabase Anda agar aplikasi dapat menyimpan dan membaca data tagihan secara online: 

-- 1. Buat Tabel Tagihan Pelanggan Perumda Tirta Indragiri
CREATE TABLE IF NOT EXISTS public.tagihan_pelanggan (
    no_register TEXT PRIMARY KEY,
    nama TEXT NOT NULL,
    alamat TEXT NOT NULL,
    periode_angka INTEGER NOT NULL DEFAULT 1,
    tagihan_air NUMERIC NOT NULL DEFAULT 0,
    tagihan_non_air NUMERIC NOT NULL DEFAULT 0,
    denda NUMERIC NOT NULL DEFAULT 0,
    tanggal_update TEXT,
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Aktifkan Row Level Security (RLS)
ALTER TABLE public.tagihan_pelanggan ENABLE ROW LEVEL SECURITY;

-- 3. Izinkan Akses Baca & Tulis Publik/Anon (Untuk Cek Tagihan & Admin Panel)
CREATE POLICY "Akses Penuh Tagihan" ON public.tagihan_pelanggan
    FOR ALL
    USING (true)
    WITH CHECK (true);

-- 4. Contoh Data Awal (Opsional)
INSERT INTO public.tagihan_pelanggan (no_register, nama, alamat, periode_angka, tagihan_air, tagihan_non_air, denda, tanggal_update)
VALUES 
('01001', 'H. MUHAMMAD YUSUF', 'JL. M. BOYA NO. 15 TEMBILAHAN', 1, 75000, 5000, 0, 'Senin, 23 November 2026'),
('01002', 'SITI RAHMAWATI', 'JL. SUDIRMAN LR. CEMPAKA NO. 08', 3, 210000, 15000, 15000, 'Senin, 23 November 2026')
ON CONFLICT (no_register) DO NOTHING;
